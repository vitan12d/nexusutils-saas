# Apache Server Rewrite Rules for SPA React routing
# INSTRUCTIONS: Rename this file to exactly ".htaccess" in the root directory where you upload your website files.
# If you are using cPanel or Hostinger File Manager, make sure to enable "Show Hidden Files" to see and edit the ".htaccess" file.

<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteBase /
  RewriteRule ^index\.html$ - [L]
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d
  RewriteRule . /index.html [L]
</IfModule>
